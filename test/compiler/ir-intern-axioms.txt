# What the compiler interning proofs state, and what they rest on.
#
# One `theorem` row per published result in formal/Common/Interning.v - every
# `Theorem` and `Corollary` it declares - in the order the file declares them,
# which is also the order the file's own `Print Assumptions` commands query
# them. Every `theorem` row is followed by exactly one `type` row and then by
# its assumption rows.
#
# There is one assumption row per result and it is always `closed`, which is
# Rocq reporting "Closed under the global context". That is the whole point of
# this file: the interning proofs rest on nothing, and the gate in
# test/compiler/ir-intern-proof.f reads this manifest with assumption rows
# FORBIDDEN, so an `axiom` row cannot be written here at all. This is stricter
# than the sibling identity manifest in test/compiler/ir-id-axioms.txt, which
# names the two assumptions of the host compare-and-swap boundary and holds the
# observed set equal to them. Here the set is empty and is checked twice: once
# when this file is read, and once against what Rocq printed.
#
# A `type` row is this file's copy of what that result states, written as Rocq
# prints the statement in the environment the generated obligations file sets
# up: Stdlib Bool, List, Lia, Arith, Permutation and Wf_nat imported, list
# notations imported, and Habu.Common.Interning imported last so a bare name
# resolves to the model's own. The parity gate turns each row into
# `Definition pinned_statement_<k> : <type row> := @<theorem name>.` in the file
# it asks Rocq to compile, so Rocq's own type checker decides whether the
# statement in the proof file still is the statement written here. It decides
# that up to conversion, so renaming a bound variable is not a change but
# weakening, strengthening, or trivialising a statement is. These rows are the
# authority: the generated file is a build output and never reads a statement
# out of the proof source, so changing the proof file alone cannot move the
# target it is checked against. The `@` makes the generic model's implicit key
# type and comparison explicit, which is why every row that quantifies over a
# key type opens with `forall (K : Type)`.
#
# Lines starting with `#`, and blank lines, are commentary. A `type` row is
# handed to Rocq. Every other row is compared byte for byte.

theorem Habu.Common.Interning.intern_idempotent
type forall (K : Type) (eqb : K -> K -> bool), (forall a : K, eqb a a = true) -> forall (t : table K) (k : K) (t' : table K) (i : nat), intern eqb t k = Some (t', i) -> intern eqb t' k = Some (t', i)
closed
theorem Habu.Common.Interning.intern_wf
type forall (K : Type) (eqb : K -> K -> bool), (forall a : K, eqb a a = true) -> forall (t : table K) (k : K) (t' : table K) (i : nat), table_wf t -> intern eqb t k = Some (t', i) -> table_wf t'
closed
theorem Habu.Common.Interning.intern_roundtrip
type forall (K : Type) (eqb : K -> K -> bool), (forall a b : K, eqb a b = true -> a = b) -> forall (t : table K) (k : K) (t' : table K) (i : nat), intern eqb t k = Some (t', i) -> key_at t' i = Some k
closed
theorem Habu.Common.Interning.intern_ordinal_in_range
type forall (K : Type) (eqb : K -> K -> bool), (forall a b : K, eqb a b = true -> a = b) -> forall (t : table K) (k : K) (t' : table K) (i : nat), intern eqb t k = Some (t', i) -> i < count t'
closed
theorem Habu.Common.Interning.key_injective
type forall (K : Type) (t : table K) (i j : nat) (k : K), table_wf t -> key_at t i = Some k -> key_at t j = Some k -> i = j
closed
theorem Habu.Common.Interning.intern_id_iff_key
type forall (K : Type) (eqb : K -> K -> bool), (forall a b : K, eqb a b = true -> a = b) -> (forall a : K, eqb a a = true) -> forall (t : table K) (k1 k2 : K) (t1 t2 : table K) (i1 i2 : nat), table_wf t -> intern eqb t k1 = Some (t1, i1) -> intern eqb t1 k2 = Some (t2, i2) -> i1 = i2 <-> k1 = k2
closed
theorem Habu.Common.Interning.intern_key_stable
type forall (K : Type) (eqb : K -> K -> bool), (forall a b : K, eqb a b = true -> a = b) -> forall (t : table K) (k : K) (t' u : table K) (i : nat), intern eqb t k = Some (t', i) -> extends t' u -> key_at u i = Some k
closed
theorem Habu.Common.Interning.intern_answer_stable
type forall (K : Type) (eqb : K -> K -> bool), (forall a : K, eqb a a = true) -> forall (t : table K) (k : K) (t' u : table K) (i : nat), intern eqb t k = Some (t', i) -> extends t' u -> intern eqb u k = Some (u, i)
closed
theorem Habu.Common.Interning.ceiling_fail_closed
type forall (K : Type) (eqb : K -> K -> bool) (t : table K) (k : K), find eqb (rows t) k = None -> ceiling t <= count t -> intern_step eqb t k = (t, None)
closed
theorem Habu.Common.Interning.order_independent
type forall (K : Type) (eqb : K -> K -> bool), (forall a b : K, eqb a b = true -> a = b) -> (forall a : K, eqb a a = true) -> forall (t0 : table K) (ks1 ks2 : list K) (u1 u2 : table K), table_wf t0 -> rows t0 = [] -> intern_all eqb t0 ks1 = Some u1 -> intern_all eqb t0 ks2 = Some u2 -> (forall x : K, In x ks1 <-> In x ks2) -> Permutation (rows u1) (rows u2)
closed
theorem Habu.Common.Interning.order_independent_permuted_input
type forall (K : Type) (eqb : K -> K -> bool), (forall a b : K, eqb a b = true -> a = b) -> (forall a : K, eqb a a = true) -> forall (t0 : table K) (ks1 ks2 : list K) (u1 u2 : table K), table_wf t0 -> rows t0 = [] -> Permutation ks1 ks2 -> intern_all eqb t0 ks1 = Some u1 -> intern_all eqb t0 ks2 = Some u2 -> Permutation (rows u1) (rows u2)
closed
